--BẢNG LƯU LỊCH SỬ CẬP NHẬT THRESHOLD
--Khi người dùng thay đổi ngưỡng, cần lưu lại ai đổi – đổi lúc nào – và đổi từ giá trị bao nhiêu sang bao nhiêu.

CREATE TABLE ThresholdUpdateLog (
    LogID INT IDENTITY(1,1) PRIMARY KEY,
    ThresholdID INT NOT NULL FOREIGN KEY REFERENCES Threshold(ThresholdID),
    UpdatedBy INT NOT NULL FOREIGN KEY REFERENCES [User](UserID),
    OldMinValue FLOAT,
    OldMaxValue FLOAT,
    OldLevelName NVARCHAR(50),
    OldColor NVARCHAR(20),
    OldAlertLevel INT,
    OldMessage NVARCHAR(300),

    NewMinValue FLOAT,
    NewMaxValue FLOAT,
    NewLevelName NVARCHAR(50),
    NewColor NVARCHAR(20),
    NewAlertLevel INT,
    NewMessage NVARCHAR(300),

    UpdatedAt DATETIME DEFAULT GETDATE()
);





--THÊM PROCEDURE ĐỂ UPDATE NGƯỠNG (có ghi log)
--Procedure này đảm bảo khi update luôn lưu lịch sử tự động.

CREATE PROCEDURE UpdateThreshold
    @ThresholdID INT,
    @LevelName NVARCHAR(50),
    @MinValue FLOAT,
    @MaxValue FLOAT,
    @AlertLevel INT,
    @Message NVARCHAR(300),
    @UpdatedBy INT
AS
BEGIN
    SET NOCOUNT ON;

    -- Kiểm tra tồn tại Threshold
    IF NOT EXISTS (SELECT 1 FROM Threshold WHERE ThresholdID = @ThresholdID)
    BEGIN
        RAISERROR(N'ThresholdID không tồn tại.', 16, 1);
        RETURN;
    END;

    -- Lấy giá trị cũ
    DECLARE 
        @OldMin FLOAT,
        @OldMax FLOAT,
        @OldLevelName NVARCHAR(50),
        @OldAlertLevel INT,
        @OldMessage NVARCHAR(300);

    SELECT 
        @OldMin = MinValue,
        @OldMax = MaxValue,
        @OldLevelName = LevelName,
        @OldAlertLevel = AlertLevel,
        @OldMessage = Message
    FROM Threshold
    WHERE ThresholdID = @ThresholdID;

    -- Ghi log
    INSERT INTO ThresholdUpdateLog (
        ThresholdID, UpdatedBy,
        OldMinValue, OldMaxValue, OldLevelName, OldAlertLevel, OldMessage,
        NewMinValue, NewMaxValue, NewLevelName, NewAlertLevel, NewMessage,
        UpdatedAt
    )
    VALUES (
        @ThresholdID, @UpdatedBy,
        @OldMin, @OldMax, @OldLevelName, @OldAlertLevel, @OldMessage,
        @MinValue, @MaxValue, @LevelName, @AlertLevel, @Message,
        GETDATE()
    );

    -- Cập nhật Threshold
    UPDATE Threshold
    SET
        LevelName = @LevelName,
        MinValue = @MinValue,
        MaxValue = @MaxValue,
        AlertLevel = @AlertLevel,
        Message = @Message
   
    WHERE ThresholdID = @ThresholdID;
END;
GO
